\newcommand{\thesischapterstyle}{%
  \titleformat{\chapter}[display]
    {\normalfont\bfseries\Huge\filleft}
    {\MakeUppercase\chaptertitlename\ \thechapter}{0pt}{{##1}}
  \titlespacing*{\chapter}{0pt}{0pt}{30pt}%
}

\titleformat{\section}{\normalfont\LARGE\bfseries}{\thesection}{1em}{#1}
\titleformat{\subsection}{\normalfont\Large\bfseries}{\thesubsection}{1em}{#1}
\titleformat{\subsubsection}{\normalfont\large\bfseries}{\thesubsubsection}{1em}{#1}
